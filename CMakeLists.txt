cmake_minimum_required(VERSION 3.22)
project(ToyCars VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(TOYCARS_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(TOYCARS_BUILD_FONT_BAKER "Build the development-only font atlas baker" OFF)
option(TOYCARS_GRAPHICS_TESTS "Build integration tests requiring a desktop display" OFF)

if(ANDROID OR CMAKE_SYSTEM_NAME STREQUAL "iOS")
  if(NOT TARGET SDL3::SDL3)
    if(DEFINED SDL3_SOURCE_DIR)
      if(ANDROID)
        set(SDL_SHARED ON CACHE BOOL "" FORCE)
        set(SDL_STATIC OFF CACHE BOOL "" FORCE)
      else()
        set(SDL_SHARED OFF CACHE BOOL "" FORCE)
        set(SDL_STATIC ON CACHE BOOL "" FORCE)
      endif()
      add_subdirectory("${SDL3_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/sdl3")
    else()
      find_package(SDL3 REQUIRED CONFIG)
    endif()
  endif()
else()
  find_package(SDL3 REQUIRED CONFIG HINTS /opt/homebrew/opt/sdl3/lib/cmake/SDL3)
endif()

set(GAME_SOURCES main.c src/game.c src/props.c src/renderer.c src/ui.c src/audio.c src/music.c src/assets.c)
if(ANDROID)
  add_library(main SHARED ${GAME_SOURCES})
  set(GAME_TARGET main)
else()
  add_executable(ToyCars ${GAME_SOURCES})
  set(GAME_TARGET ToyCars)
endif()
target_include_directories(${GAME_TARGET} PRIVATE src)
target_link_libraries(${GAME_TARGET} PRIVATE SDL3::SDL3)
target_compile_options(${GAME_TARGET} PRIVATE -Wall -Wextra -Wpedantic)
if(APPLE AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  target_compile_definitions(${GAME_TARGET} PRIVATE GL_SILENCE_DEPRECATION)
  target_link_libraries(${GAME_TARGET} PRIVATE "-framework OpenGL")
elseif(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/platform/ios/Info.plist")
  target_link_libraries(${GAME_TARGET} PRIVATE "-framework OpenGLES")
  set_target_properties(${GAME_TARGET} PROPERTIES MACOSX_BUNDLE TRUE
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_SOURCE_DIR}/platform/ios/Info.plist"
    XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER "com.toycars.game"
    XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1,2")
  file(GLOB_RECURSE GAME_ASSETS "${CMAKE_SOURCE_DIR}/assets/*")
  target_sources(${GAME_TARGET} PRIVATE ${GAME_ASSETS})
  foreach(ASSET ${GAME_ASSETS})
    file(RELATIVE_PATH REL "${CMAKE_SOURCE_DIR}" "${ASSET}")
    get_filename_component(DIR "${REL}" DIRECTORY)
    set_source_files_properties("${ASSET}" PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/${DIR}")
  endforeach()
elseif(ANDROID)
  target_link_libraries(${GAME_TARGET} PRIVATE GLESv3 m log)
else()
  find_package(OpenGL REQUIRED)
  target_link_libraries(${GAME_TARGET} PRIVATE OpenGL::GL m)
endif()
if(TOYCARS_SANITIZE)
  target_compile_options(${GAME_TARGET} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(${GAME_TARGET} PRIVATE -fsanitize=address,undefined)
endif()
if(NOT ANDROID AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  # Assets can change without any C source changing. Run the copy for every
  # build, including an explicit --target ToyCars, without forcing a relink.
  add_custom_target(toycars_assets
    COMMAND ${CMAKE_COMMAND} -E copy_directory
      "${CMAKE_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:ToyCars>/assets"
    COMMENT "Updating ToyCars assets"
    VERBATIM)
  add_dependencies(ToyCars toycars_assets)
  if(TOYCARS_GRAPHICS_TESTS)
    # The test includes main.c to exercise the actual event loop and assets.c
    # to redirect preferences into an isolated fixture.
    set(GRAPHICS_TEST_SOURCES ${GAME_SOURCES})
    list(REMOVE_ITEM GRAPHICS_TEST_SOURCES main.c src/assets.c)
    add_executable(ToyCarsGraphicsTests tests/graphics.c ${GRAPHICS_TEST_SOURCES})
    foreach(PROPERTY INCLUDE_DIRECTORIES COMPILE_OPTIONS COMPILE_DEFINITIONS LINK_LIBRARIES LINK_OPTIONS)
      get_target_property(VALUE ToyCars ${PROPERTY})
      if(VALUE)
        set_property(TARGET ToyCarsGraphicsTests PROPERTY ${PROPERTY} "${VALUE}")
      endif()
    endforeach()
    add_dependencies(ToyCarsGraphicsTests toycars_assets)
  endif()
  enable_testing()
  add_executable(ToyCarsProgressionTests tests/progression.c src/game.c src/props.c)
  foreach(PROPERTY INCLUDE_DIRECTORIES COMPILE_OPTIONS COMPILE_DEFINITIONS LINK_LIBRARIES LINK_OPTIONS)
    get_target_property(VALUE ToyCars ${PROPERTY})
    if(VALUE)
      set_property(TARGET ToyCarsProgressionTests PROPERTY ${PROPERTY} "${VALUE}")
    endif()
  endforeach()
  add_dependencies(ToyCarsProgressionTests toycars_assets)
  add_test(NAME progression COMMAND ToyCarsProgressionTests "${CMAKE_BINARY_DIR}/progression-test-profile")
  set_tests_properties(progression PROPERTIES WORKING_DIRECTORY "${CMAKE_BINARY_DIR}")
  add_executable(ToyCarsMusicTests tests/music.c src/music.c)
  target_include_directories(ToyCarsMusicTests PRIVATE src)
  target_compile_options(ToyCarsMusicTests PRIVATE -Wall -Wextra -Wpedantic)
  if(NOT APPLE)
    target_link_libraries(ToyCarsMusicTests PRIVATE m)
  endif()
  if(TOYCARS_SANITIZE)
    target_compile_options(ToyCarsMusicTests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(ToyCarsMusicTests PRIVATE -fsanitize=address,undefined)
  endif()
  add_executable(ToyCarsPropsTests tests/props.c src/props.c src/game.c src/assets.c)
  target_include_directories(ToyCarsPropsTests PRIVATE src)
  target_link_libraries(ToyCarsPropsTests PRIVATE SDL3::SDL3)
  target_compile_options(ToyCarsPropsTests PRIVATE -Wall -Wextra -Wpedantic)
  if(NOT APPLE)
    target_link_libraries(ToyCarsPropsTests PRIVATE m)
  endif()
  if(TOYCARS_SANITIZE)
    target_compile_options(ToyCarsPropsTests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(ToyCarsPropsTests PRIVATE -fsanitize=address,undefined)
  endif()
  add_test(NAME props COMMAND ToyCarsPropsTests)
  add_test(NAME music COMMAND ToyCarsMusicTests)
  add_test(NAME simulation COMMAND ToyCars --test)
  set_tests_properties(simulation PROPERTIES WORKING_DIRECTORY "${CMAKE_BINARY_DIR}")
  add_test(NAME asset_updates COMMAND ${CMAKE_COMMAND}
    "-DPROJECT_SOURCE_DIR=${CMAKE_SOURCE_DIR}"
    "-DTEST_BINARY_DIR=${CMAKE_BINARY_DIR}/asset-update-test"
    "-DTEST_GENERATOR=${CMAKE_GENERATOR}"
    "-DTEST_CONFIG=$<CONFIG>"
    "-DTEST_C_COMPILER=${CMAKE_C_COMPILER}"
    "-DTEST_SDL3_DIR=${SDL3_DIR}"
    -P "${CMAKE_SOURCE_DIR}/tests/asset_updates.cmake")
  if(TOYCARS_GRAPHICS_TESTS)
    add_test(NAME graphics COMMAND ToyCarsGraphicsTests "${CMAKE_BINARY_DIR}/graphics-test-profile")
    set_tests_properties(graphics PROPERTIES LABELS graphics WORKING_DIRECTORY "${CMAKE_BINARY_DIR}")
  endif()
endif()
if(TOYCARS_BUILD_FONT_BAKER)
  find_package(SDL3_ttf REQUIRED CONFIG HINTS /opt/homebrew/opt/sdl3_ttf/lib/cmake/SDL3_ttf)
  add_executable(bake_fonts tools/bake_fonts.c)
  target_link_libraries(bake_fonts PRIVATE SDL3::SDL3 SDL3_ttf::SDL3_ttf)
endif()
